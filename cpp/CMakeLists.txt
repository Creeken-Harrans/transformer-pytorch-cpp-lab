cmake_minimum_required(VERSION 3.16)
project(transformer_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(TLAB_ENABLE_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
add_library(tlab src/tensor.cpp src/model.cpp src/format.cpp)
target_include_directories(tlab PUBLIC include)
target_compile_options(tlab PRIVATE -Wall -Wextra -Wpedantic)
if(TLAB_ENABLE_SANITIZERS)
  target_compile_options(tlab PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(tlab PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(tlab_tests tests/test_main.cpp)
target_link_libraries(tlab_tests PRIVATE tlab)
target_compile_options(tlab_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME core_math COMMAND tlab_tests)
add_executable(tlab_model_tests tests/test_model.cpp)
target_link_libraries(tlab_model_tests PRIVATE tlab)
target_compile_options(tlab_model_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME model_format COMMAND tlab_model_tests)
add_executable(transformer_cpp src/main.cpp)
target_link_libraries(transformer_cpp PRIVATE tlab)
target_compile_options(transformer_cpp PRIVATE -Wall -Wextra -Wpedantic)
add_executable(tlab_fixture tests/fixture.cpp)
target_link_libraries(tlab_fixture PRIVATE tlab)
target_compile_options(tlab_fixture PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME cli_contract COMMAND ${CMAKE_COMMAND}
  -DCLI=$<TARGET_FILE:transformer_cpp> -DFIXTURE=$<TARGET_FILE:tlab_fixture>
  -DDIR=${CMAKE_CURRENT_BINARY_DIR}/cli-test -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
